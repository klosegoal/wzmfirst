class Solution {
public:
    int romanToInt(string s) {
        int mp[256]={0};
        mp['I']=1;
        mp['V']=5;
        mp['X']=10;
        mp['L']=50;
        mp['C']=100;
        mp['D']=500;
        mp['M']=1000;
        int res=0;
        int n=s.length();
        for(int i=0;i<n;i++){
            if(i<n-1&&mp[s[i+1]]>mp[s[i]]){
                res-=mp[s[i]];
            }else{
                res+=mp[s[i]];
            }
            
        }
        return res;
    }
};
