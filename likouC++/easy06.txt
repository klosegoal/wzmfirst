class Solution {
public:
    int lengthOfLastWord(string s) {
        int index=s.size()-1;
        while (s[index]==' '){
            index--;
        }
        int wordlength=0;
        while(s[index]!=' '&& index>=0){
            
            wordlength++;
            index--;
        }
        return wordlength;
    }
};
